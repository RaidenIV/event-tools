function loadCSV(fileOrText) { const applyText = (text) => { resetForm(); const lines = String(text || "").split(/\r?\n/); const parseLabelValue = (line) => { const idx = line.indexOf(","); if (idx < 0) return [line.trim(), ""]; const label = line.slice(0, idx).trim(); let value = line.slice(idx + 1).trim(); if (value.startsWith('"') && value.endsWith('"')) { value = value.slice(1, -1).replace(/""/g, '"'); } return [label, value]; }; const setVal = (id, value) => { const el = document.getElementById(id); if (el) el.value = value; }; // PASS 1: read pairs and determine repeater sizes const pairs = []; let maxHead = 1, maxLocal = 0, maxCDJ = 0, maxRunner = 0, maxVendor = 0, maxOther = 0; const otherCounts = {}; let legacyFbAmount = "", legacyFbAcct = ""; let legacyIgAmount = "", legacyIgAcct = ""; for (const raw of lines) { if (!raw || !raw.trim()) continue; const [label, value] = parseLabelValue(raw); pairs.push([label, value]); if (/^Headliner\s+(\d+)\s+/.test(label)) { maxHead = Math.max(maxHead, parseInt(label.match(/^Headliner\s+(\d+)/)[1], 10)); } if (/^Local DJ\s+(\d+)\s+/.test(label)) { maxLocal = Math.max(maxLocal, parseInt(label.match(/^Local DJ\s+(\d+)/)[1], 10)); } if (/^CDJ\s+(\d+)\s+Fee$/.test(label)) { maxCDJ = Math.max(maxCDJ, parseInt(label.match(/^CDJ\s+(\d+)/)[1], 10)); } if (/^Show Runner\s+(\d+)\s+Fee$/.test(label)) { maxRunner = Math.max(maxRunner, parseInt(label.match(/^Show Runner\s+(\d+)/)[1], 10)); } if (/^Vendor\s+(\d+)\s+/.test(label)) { maxVendor = Math.max(maxVendor, parseInt(label.match(/^Vendor\s+(\d+)/)[1], 10)); } if (/^Category\s+(\d+)\s+Items$/.test(label)) { const c = parseInt(label.match(/^Category\s+(\d+)/)[1], 10); maxOther = Math.max(maxOther, c); otherCounts[c] = parseInt(value, 10) || 0; } if (label === "Facebook Ads") legacyFbAmount = value; if (label === "Facebook Ads Account") legacyFbAcct = value; if (label === "Instagram Ads") legacyIgAmount = value; if (label === "Instagram Ads Account") legacyIgAcct = value; } // Resize repeaters setVal("numHeadliners", String(maxHead)); regenerateHeadliners?.(); setVal("numLocalDJs", String(maxLocal)); regenerateLocalDJs?.(); setVal("numCDJs", String(maxCDJ)); regenerateCDJs?.(); setVal("numShowRunners", String(maxRunner)); regenerateShowRunners?.(); setVal("numMerchVendors", String(maxVendor)); regenerateVendors?.(); setVal("numOtherCategories", String(maxOther)); regenerateOtherCategories?.(); // Create Other items counts before setting item values for (let c = 1; c <= maxOther; c++) { setVal(`otherCategoryCount_${c}`, String(otherCounts[c] || 0)); regenerateOtherItems?.(c); } // PASS 2: assign values let currentOtherCategory = 0; for (const [label, value] of pairs) { if (label === "Show Title") { setVal("showTitle", value); continue; } if (label === "Show Date") { setVal("showDate", value); continue; } if (label.startsWith("Headliner ")) { const idx = parseInt(label.split(" ")[1], 10); if (label.includes("Fee")) setVal(`headliner_fee_${idx}`, value); if (label.includes("Hotel")) setVal(`headliner_hotel_${idx}`, value); if (label.includes("Rider")) setVal(`headliner_rider_${idx}`, value); if (label.includes("Name")) setVal(`headliner_name_${idx}`, value); continue; } if (label === "Direct Support Fee") { setVal("directSupport", value); continue; } if (label.startsWith("Local DJ")) { const idx = parseInt(label.split(" ")[2], 10); if (label.includes("Name")) setVal(`localDJ_name_${idx}`, value); if (label.includes("Fee")) setVal(`localDJ_fee_${idx}`, value); continue; } if (label === "VJ Fee") { setVal("vjFee", value); continue; } if (label === "Venue") { setVal("venue", value); continue; } if (label === "LED Wall") { setVal("ledWall", value); continue; } if (label === "Lights") { setVal("lights", value); continue; } if (label === "Lasers") { setVal("lasers", value); continue; } if (/^CDJ\s+\d+\s+Fee$/.test(label)) { const idx = parseInt(label.split(" ")[1], 10); setVal(`cdj_fee_${idx}`, value); continue; } if (label === "Sound") { setVal("sound", value); continue; } if (label === "Mixer") { setVal("mixer", value); continue; } if (label === "Table") { setVal("table", value); continue; } // MARKETING (new split labels) if (label === "Facebook Ads XODIA") { setVal("facebookAdsXodia", value); continue; } if (label === "Facebook Ads SPACE CAMP HQ") { setVal("facebookAdsSpaceCampHQ", value); continue; } if (label === "Instagram Ads XODIA") { setVal("instagramAdsXodia", value); continue; } if (label === "Instagram Ads SPACE CAMP HQ") { setVal("instagramAdsSpaceCampHQ", value); continue; } // ✅ Physical Flyers import if (label === "Physical Flyers") { setVal("physicalFlyers", value); setVal("flyerCost", value); continue; } if (label === "Eventbrite Ads") { setVal("eventbriteAds", value); continue; } // MARKETING (legacy) if (label === "Facebook Ads") { legacyFbAmount = value; continue; } if (label === "Facebook Ads Account") { legacyFbAcct = value; continue; } if (label === "Instagram Ads") { legacyIgAmount = value; continue; } if (label === "Instagram Ads Account") { legacyIgAcct = value; continue; } if (label === "Door Staff") { setVal("doorStaff", value); continue; } if (label === "Merch Table") { setVal("merchTable", value); continue; } if (label === "Transportation") { setVal("transportation", value); continue; } if (label.startsWith("Show Runner") && label.includes("Fee")) { const idx = parseInt(label.split(" ")[2], 10); setVal(`showRunner_fee_${idx}`, value); continue; } if (label.startsWith("Category") && label.includes("Name")) { currentOtherCategory = parseInt(label.split(" ")[1], 10); setVal(`otherCategoryName_${currentOtherCategory}`, value); continue; } if (label.startsWith("Category") && label.includes("Items")) { const c = parseInt(label.split(" ")[1], 10); setVal(`otherCategoryCount_${c}`, value); regenerateOtherItems?.(c); continue; } if (label.startsWith("Item") && label.includes("Name")) { const idx = parseInt(label.split(" ")[1], 10); setVal(`otherCategory_${currentOtherCategory}_itemName_${idx}`, value); continue; } if (label.startsWith("Item") && label.includes("Fee")) { const idx = parseInt(label.split(" ")[1], 10); setVal(`otherCategory_${currentOtherCategory}_itemFee_${idx}`, value); continue; } if (label === "Eventbrite Sales") { setVal("eventbriteSales", value); continue; } if (label === "DJ Presales") { setVal("djPresales", value); continue; } if (label === "Promo Team") { setVal("promoTeam", value); continue; } if (label === "Door Sales") { setVal("doorSales", value); continue; } if (label === "Merch Sold") { setVal("merchSold", value); continue; } if (label.startsWith("Vendor") && label.includes("Name")) { const idx = parseInt(label.split(" ")[1], 10); setVal(`merchVendor_name_${idx}`, value); continue; } if (label.startsWith("Vendor") && label.includes("Fee")) { const idx = parseInt(label.split(" ")[1], 10); setVal(`merchVendor_fee_${idx}`, value); continue; } } // Apply legacy marketing into split fields if split inputs exist const hasFbSplit = !!document.getElementById("facebookAdsXodia") || !!document.getElementById("facebookAdsSpaceCampHQ"); const hasIgSplit = !!document.getElementById("instagramAdsXodia") || !!document.getElementById("instagramAdsSpaceCampHQ"); if (hasFbSplit && legacyFbAmount) { if ((legacyFbAcct || "").toUpperCase().includes("SPACE")) { setVal("facebookAdsSpaceCampHQ", legacyFbAmount); } else { setVal("facebookAdsXodia", legacyFbAmount); } } if (hasIgSplit && legacyIgAmount) { if ((legacyIgAcct || "").toUpperCase().includes("SPACE")) { setVal("instagramAdsSpaceCampHQ", legacyIgAmount); } else { setVal("instagramAdsXodia", legacyIgAmount); } } updateBudget?.(); }; // If passed a File, read it; else treat as CSV text if (fileOrText && typeof File !== "undefined" && fileOrText instanceof File) { const reader = new FileReader(); reader.onload = (ev) => applyText(ev.target.result); reader.readAsText(fileOrText); } else { applyText(fileOrText); } }
